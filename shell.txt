#!/bin/bash

# Global configuration
JAHIA_HOME="/appli/jahia"
TOMCAT_HOME="$JAHIA_HOME/tomcat"
LOG_DIR="$TOMCAT_HOME/logs"
MIN_DISK_SPACE_GB=20

# Jahia versions
JAHIA_CURRENT_VERSION="8.1.6.1"
JAHIA_TARGET_VERSION="8.2.3.0"

# Patch configuration
PATCH_DIR="/appli/patches"
PATCH_JAR="Jahia-EnterpriseDistribution-Upgrade-${JAHIA_CURRENT_VERSION}_to_${JAHIA_TARGET_VERSION}.jar"

clean_old_tomcat_logs() {
    local log_dir="$JAHIA_HOME/tomcat/logs"
    local errors_dir
    local dir_name

    echo "Cleaning Tomcat logs older than 30 days..."

    if [[ ! -d "$log_dir" ]]; then
        echo "ERROR: Log directory not found: $log_dir"
        return 1
    fi

    # Delete dated log files older than 30 days
    find "$log_dir" -maxdepth 1 -type f \
        -regextype posix-extended \
        -regex '.*/(access-log|catalina|host-manager|jahia|jahia_access|localhost)[._][0-9]{4}-[0-9]{2}-[0-9]{2}[.](txt|log)' \
        -mtime +30 \
        -print \
        -delete

    # Delete dated directories older than 30 days
    for dir_name in jahia-errors jahia-heaps jahia-threads; do
        errors_dir="$log_dir/$dir_name"

        if [[ -d "$errors_dir" ]]; then
            find "$errors_dir" -mindepth 1 -maxdepth 1 -type d \
                -regextype posix-extended \
                -regex '.*/[0-9]{4}_[0-9]{2}_[0-9]{2}' \
                -mtime +30 \
                -print \
                -exec rm -rf {} +
        fi
    done

    echo "Tomcat log cleanup completed."
}


check_disk_space() {
    local required_gb="${1:-$MIN_DISK_SPACE_GB}"
    local check_path="${2:-$JAHIA_HOME}"
    local available_kb
    local required_kb=$((required_gb * 1024 * 1024))

    echo "Checking available disk space..."

    available_kb=$(df -Pk "$check_path" | awk 'NR==2 {print $4}')

    if [[ ! "$available_kb" =~ ^[0-9]+$ ]]; then
        echo "ERROR: Unable to determine available disk space."
        return 1
    fi

    echo "Available: $((available_kb / 1024 / 1024)) GiB"
    echo "Required: ${required_gb} GiB"

    if (( available_kb < required_kb )); then
        echo "ERROR: Insufficient disk space."
        return 1
    fi

    echo "Disk space check passed."
    return 0
}

if ! check_disk_space "$JAHIA_HOME"; then
    echo "Migration aborted."
    exit 1
fi

apply_jahia_patch() {
    local patch_jar="$1"

    echo "Starting Jahia patch..."

    if [[ ! -f "$patch_jar" ]]; then
        echo "ERROR: Patch file not found: $patch_jar"
        return 1
    fi

    java -jar "$patch_jar" -console <<EOF
A
A
A
EOF

    local exit_code=$?

    if [[ $exit_code -ne 0 ]]; then
        echo "ERROR: Jahia patch failed (exit code: $exit_code)."
        return 1
    fi

    echo "Jahia patch completed successfully."
    return 0
}


clean_tomcat_temp() {
    echo "Cleaning Tomcat temporary directories..."

    local dir

    for dir in "$TOMCAT_HOME/temp" "$TOMCAT_HOME/work"; do
        if [[ -d "$dir" ]]; then
            echo "Cleaning $dir..."
            rm -rf -- "$dir" || {
                echo "ERROR: Unable to clean $dir"
                return 1
            }
        else
            echo "Directory not found, skipping: $dir"
        fi
    done

    echo "Tomcat temporary directories cleaned successfully."
}

# Clean Tomcat temporary directories
if ! clean_tomcat_temp; then
    echo "Cleanup failed. Migration aborted."
    exit 1
fi
